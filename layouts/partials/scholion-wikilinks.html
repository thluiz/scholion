{{/* Resolve wikilinks [[slug]] e [[slug|texto]] no corpo já renderizado. Hugo não tem suporte nativo a wikilinks; sem isto, [[...]] sai como texto cru. - [[slug]] → Título da nota - [[slug|texto]] → texto Procura o alvo em /notes/ e, em fallback, em /research/. Alvo inexistente vira para ficar visível. Recebe a Page como contexto (.). */}} {{- $content := .Content -}} {{- range $m := (findRE `\[\[[^\]]+\]\]` $content) | uniq -}} {{- $inner := strings.TrimSuffix "]]" (strings.TrimPrefix "[[" $m) -}} {{- $parts := split $inner "|" -}} {{- $slug := trim (index $parts 0) " " -}} {{- $text := cond (gt (len $parts) 1) (trim (index $parts 1) " ") "" -}} {{- $target := or (site.GetPage (printf "/notes/%s" $slug)) (site.GetPage (printf "/research/%s" $slug)) -}} {{- if $target -}} {{- $label := cond (ne $text "") $text $target.Title -}} {{- $a := printf `%s` $target.RelPermalink $label -}} {{- $content = replace $content $m $a -}} {{- else -}} {{- $label := cond (ne $text "") $text $slug -}} {{- $content = replace $content $m (printf `%s` $slug $label) -}} {{- end -}} {{- end -}} {{ $content | safeHTML }}
